// Research-only worker. This is not loaded by the application or Pages build.
// Observe actual Wasm linear memories without relying on private SDK fields.
const memories = new Set(), NativeMemory = WebAssembly.Memory;
WebAssembly.Memory = class extends NativeMemory { constructor(...args) { super(...args); memories.add(this); } };
for (const method of ["instantiate", "instantiateStreaming"]) {
  const original = WebAssembly[method];
  WebAssembly[method] = async function (...args) {
    for (const namespace of Object.values(args[1] ?? {})) for (const value of Object.values(namespace)) if (value instanceof NativeMemory) memories.add(value);
    const result = await original.apply(WebAssembly, args);
    for (const value of Object.values((result.instance ?? result).exports)) if (value instanceof NativeMemory) memories.add(value);
    return result;
  };
}
const wasmBytes = () => [...memories].reduce((sum, memory) => sum + memory.buffer.byteLength, 0);
const hash = async (data) => Array.from(new Uint8Array(await crypto.subtle.digest("SHA-256", data)), (v) => v.toString(16).padStart(2, "0")).join("");
let task;
const policyViolations = [];
addEventListener("securitypolicyviolation", (event) => policyViolations.push({ blockedURI: event.blockedURI, directive: event.effectiveDirective, disposition: event.disposition }));
onmessage = async ({ data }) => {
  const reports = [], timeline = [];
  try {
    if (data.noWebGL) { const getContext = OffscreenCanvas.prototype.getContext; OffscreenCanvas.prototype.getContext = function (type, ...args) { return String(type).includes("webgl") ? null : getContext.call(this, type, ...args); }; }
    const started = performance.now(); importScripts("/sdk/vision_bundle.js");
    const bundleMs = performance.now() - started;
    const fileset = data.noSimd ? { wasmLoaderPath: "/sdk/wasm/vision_wasm_nosimd_internal.js", wasmBinaryPath: "/sdk/wasm/vision_wasm_nosimd_internal.wasm" }
      : await Vision.FilesetResolver.forVisionTasks("/sdk/wasm");
    const downloadStart = performance.now(), model = new Uint8Array(await (await fetch(`/assets/${data.file}`)).arrayBuffer());
    if (await hash(model) !== data.sha256) throw new Error("Model identity mismatch");
    const modelReadMs = performance.now() - downloadStart, initStart = performance.now();
    const canvas = new OffscreenCanvas(1, 1), options = { baseOptions: { modelAssetBuffer: model, delegate: data.delegate }, canvas };
    task = data.interactive ? await Vision.InteractiveSegmenter.createFromOptions(fileset, options)
      : await Vision.ImageSegmenter.createFromOptions(fileset, { ...options, runningMode: "IMAGE", outputConfidenceMasks: true, outputCategoryMask: false });
    const initMs = performance.now() - initStart, labels = task.getLabels?.() ?? null;
    timeline.push({ stage: "ready", wasmBytes: wasmBytes() });
    let renderer = null; const gl = canvas.getContext("webgl2");
    if (gl) { const ext = gl.getExtension("WEBGL_debug_renderer_info"); renderer = ext ? gl.getParameter(ext.UNMASKED_RENDERER_WEBGL) : gl.getParameter(gl.RENDERER); }
    for (const input of data.inputs) {
      const blob = await (await fetch(`/assets/${input.file}`)).blob(), decodeStart = performance.now(), bitmap = await createImageBitmap(blob);
      const decodeMs = performance.now() - decodeStart, timings = []; let output, maskWidth, maskHeight, channels, ranges;
      let encodeMs = null;
      if (data.interactive) { const start = performance.now(); task.setImage(bitmap); encodeMs = performance.now() - start; }
      for (let run = 0; run < data.repetitions; run++) {
        const start = performance.now();
        const consume = (masks) => {
          channels = masks.length; maskWidth = masks[0].width; maskHeight = masks[0].height;
          const arrays = masks.map((mask) => mask.getAsFloat32Array());
          ranges = arrays.map((array) => { let min = Infinity, max = -Infinity, sum = 0; for (const value of array) { min = Math.min(min, value); max = Math.max(max, value); sum += value; }
            return { min, max, mean: sum / array.length }; });
          if (ranges.some((range) => !Number.isFinite(range.mean) || range.min < -1e-5 || range.max > 1.00001)) throw new Error("Output is not a finite probability mask");
          if (!data.interactive && channels !== (data.file === "selfie.tflite" ? 1 : 6)) throw new Error("Unexpected confidence channels");
          output = new Uint8Array(maskWidth * maskHeight);
          for (let i = 0; i < output.length; i++) output[i] = Math.round(Math.max(0, Math.min(1, channels === 6 ? 1 - arrays[0][i] : arrays[0][i])) * 255);
        };
        // SDK 1.0.1 declares BrushMode in vision.d.ts but its classic bundle
        // does not export the enum object. POSITIVE is the declared value 1.
        if (data.interactive) { const mask = task.segment([{ brushMode: 1, point: [input.point], isCompleted: true }]);
          try { consume([mask]); } finally { mask.close(); }
        } else task.segment(bitmap, (result) => consume(result.confidenceMasks));
        timings.push(performance.now() - start); timeline.push({ stage: `${input.file}:${run}`, wasmBytes: wasmBytes() });
      }
      reports.push({ file: input.file, width: bitmap.width, height: bitmap.height, decodeMs, encodeMs, timings, maskWidth, maskHeight, channels, ranges,
        maskHash: await hash(output), mask: output.buffer }); bitmap.close();
    }
    task.close(); task = null; timeline.push({ stage: "closed", wasmBytes: wasmBytes() });
    // Allow policy-violation events queued by the SDK's close/flush to arrive.
    await new Promise((done) => setTimeout(done, 0));
    postMessage({ type: "result", model: data.file, delegate: data.delegate, noSimd: Boolean(data.noSimd), noWebGL: Boolean(data.noWebGL), bundleMs, modelReadMs, initMs, labels, renderer, fileset, timeline, policyViolations, reports }, reports.map((report) => report.mask));
  } catch (error) { postMessage({ type: "failure", model: data.file, delegate: data.delegate, message: error.message, stack: error.stack, timeline, policyViolations }); }
  finally { task?.close(); task = null; }
};
